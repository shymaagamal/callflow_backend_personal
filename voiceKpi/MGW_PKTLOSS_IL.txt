SELECT
   D1."DATETIME_ID" AS "DATETIMEID",
   D1."HOUR_ID" AS "HOURID",
   D1."IpNetwork" AS "IPNETWORK",
   D1."MGW" AS "MGW",
   D1."MIN_ID" AS "MINID",
   D1."MOID" AS "MOID",
   D1."pmRtpLostPkts" AS "PMRTPLOSTPKTS",
   D1."pmRtpReceivedPktsHi" AS "PMRTPRECEIVEDPKTSHI",
   D1."pmRtpReceivedPktsLo" AS "PMRTPRECEIVEDPKTSLO",
   D1."RemoteSite" AS "REMOTESITE",
   D1."SN" AS "SN"
FROM
   "DC_E_MGW_REMOTESITE_RAW" D1
WHERE
   ( D1."DATETIME_ID">= dateadd(hour, -1, getdate())
