SELECT
   D1."DATETIME_ID" AS "DATETIMEID",
   D1."ELEM" AS "ELEM",
   D1."HOUR_ID" AS "HOURID",
   D1."LOAS_ACCLOAD" AS "LOASACCLOAD",
   D1."LOAS_NSCAN" AS "LOASNSCAN",
   D1."MIN_ID" AS "MINID",
   D1."SN" AS "SN"
FROM
   "DC_E_CMN_STS_TOT_RAW" D1


WHERE
   ( D1."DATETIME_ID">= dateadd(hour, -1, getdate())
